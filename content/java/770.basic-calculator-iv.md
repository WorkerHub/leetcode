## 题目描述

给定一个表达式 `expression`，如 `"e + 8 - a + 5"` 和一个赋值映射 `evalvars = ["e"]` 和 `evalints = [1]`，返回表示简化表达式的标记列表，例如 `["-1*a","14"]`。

## 解题思路

使用多项式表示，支持加法、减法、乘法和变量替换。

## Java 代码实现

```java
import java.util.*;

class Solution {
    // 多项式用 Map 表示：系数映射（变量组合 -> 系数），常数项特殊处理
    // 实现较复杂，涉及多项式的加、减、乘运算

    public List<String> basicCalculatorIV(String expression, String[] evalvars, int[] evalints) {
        // 建立变量到值的映射
        Map<String, Integer> varMap = new HashMap<>();
        for (int i = 0; i < evalvars.length; i++) {
            varMap.put(evalvars[i], evalints[i]);
        }

        // 解析并计算表达式
        Poly result = parse(expression, varMap, new int[]{0});
        return result.toList();
    }

    // 多项式类：表示含变量的多项式
    private static class Poly {
        // 键为变量组合（如 "a*b"），值为系数；空字符串表示常数项
        Map<String, Integer> terms = new TreeMap<>();

        void add(String key, int val) {
            terms.merge(key, val, Integer::sum);
            if (terms.get(key) == 0) terms.remove(key);
        }

        // 多项式加法
        Poly plus(Poly other) {
            Poly result = new Poly();
            result.terms.putAll(terms);
            other.terms.forEach((k, v) -> result.add(k, v));
            return result;
        }

        // 多项式减法
        Poly minus(Poly other) {
            Poly result = new Poly();
            result.terms.putAll(terms);
            other.terms.forEach((k, v) -> result.add(k, -v));
            return result;
        }

        // 多项式乘法
        Poly multiply(Poly other) {
            Poly result = new Poly();
            for (Map.Entry<String, Integer> e1 : terms.entrySet()) {
                for (Map.Entry<String, Integer> e2 : other.terms.entrySet()) {
                    // 合并变量键
                    String newKey;
                    if (e1.getKey().isEmpty()) newKey = e2.getKey();
                    else if (e2.getKey().isEmpty()) newKey = e1.getKey();
                    else {
                        List<String> parts = new ArrayList<>(Arrays.asList(e1.getKey().split("\\*")));
                        parts.addAll(Arrays.asList(e2.getKey().split("\\*")));
                        Collections.sort(parts);
                        newKey = String.join("*", parts);
                    }
                    result.add(newKey, e1.getValue() * e2.getValue());
                }
            }
            return result;
        }

        // 输出格式化的列表
        List<String> toList() {
            List<String> result = new ArrayList<>();
            // 按变量度数降序、字典序排列
            List<Map.Entry<String, Integer>> entries = new ArrayList<>(terms.entrySet());
            entries.sort((a, b) -> {
                int da = a.getKey().isEmpty() ? 0 : a.getKey().split("\\*").length;
                int db = b.getKey().isEmpty() ? 0 : b.getKey().split("\\*").length;
                if (da != db) return db - da;
                return a.getKey().compareTo(b.getKey());
            });
            for (Map.Entry<String, Integer> e : entries) {
                if (e.getKey().isEmpty()) {
                    result.add(String.valueOf(e.getValue()));
                } else {
                    result.add(e.getValue() + "*" + e.getKey());
                }
            }
            return result;
        }
    }

    // 递归解析表达式
```
