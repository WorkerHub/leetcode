## 题目描述

在 `rows x cols` 网格中，猫（`C`）和鼠（`M`）轮流移动（鼠先），每次可向四方向跳最多 `mouseJump`（鼠）或 `catJump`（猫）步，不可越过墙（`#`）。鼠到食物（`F`）则鼠赢，猫到食物或与鼠同格则猫赢，步数超限则猫赢。返回鼠是否能赢。

## 解题思路

博弈记忆化搜索：状态为 `(鼠行, 鼠列, 猫行, 猫列, 轮次)`，轮次限制为 `rows*cols*2` 防止死循环。鼠回合枚举所有移动，有任意一步赢则鼠赢；猫回合枚举所有移动，有任意一步赢则猫赢。

## Java 代码实现

```java
import java.util.Arrays;

class Solution {
    private int[][] memo;
    private String[] grid;
    private int rows, cols, catJump, mouseJump;
    private int fr, fc; // 食物位置
    private static final int[] DR = {0, 0, 1, -1};
    private static final int[] DC = {1, -1, 0, 0};

    public boolean canMouseWin(String[] grid, int catJump, int mouseJump) {
        this.grid = grid;
        this.rows = grid.length;
        this.cols = grid[0].length();
        this.catJump = catJump;
        this.mouseJump = mouseJump;

        int mr = 0, mc = 0, cr = 0, cc = 0;
        for (int r = 0; r < rows; r++) {
            for (int c = 0; c < cols; c++) {
                char ch = grid[r].charAt(c);
                if (ch == 'F') { fr = r; fc = c; }
                else if (ch == 'C') { cr = r; cc = c; }
                else if (ch == 'M') { mr = r; mc = c; }
            }
        }

        int maxTurns = rows * cols * 2;
        // 状态维度：鼠位置 * 猫位置 * 轮次
        memo = new int[rows * cols * rows * cols * maxTurns + 1][1];
        int size = rows * cols * rows * cols * (maxTurns + 1);
        memo = null;
        // 用 int[] 4D 数组记忆化
        int[][][][] dp = new int[rows][cols][rows * cols][maxTurns + 1];
        // 0=未计算, 1=鼠赢, 2=猫赢
        for (int[][][] a : dp) for (int[][] b : a) for (int[] c2 : b) Arrays.fill(c2, -1);

        return solve(dp, mr, mc, cr * cols + cc, 0, maxTurns) == 1;
    }

    private int solve(int[][][][] dp, int mr, int mc, int catIdx, int turn, int maxTurns) {
        int cr = catIdx / cols, cc = catIdx % cols;
        // 步数超限，猫赢
        if (turn >= maxTurns) return 2;
        int cached = dp[mr][mc][catIdx][turn];
        if (cached != -1) return cached;

        int result;
        if (turn % 2 == 0) {
            // 鼠的回合：有任意一步鼠赢则鼠赢，否则猫赢
            result = 2;
            outer:
            for (int d = 0; d < 4; d++) {
                // 原地不动
                if (d == 0) {
                    // 跳 0 步
                }
                for (int step = 0; step <= mouseJump; step++) {
                    int nr = mr + DR[d] * step, nc = mc + DC[d] * step;
                    if (nr < 0 || nr >= rows || nc < 0 || nc >= cols) break;
                    if (grid[nr].charAt(nc) == '#') break;
                    if (nr == fr && nc == fc) { result = 1; break outer; } // 鼠到食物
                    if (nr == cr && nc == cc) continue; // 鼠与猫同格，跳过
                    if (solve(dp, nr, nc, catIdx, turn + 1, maxTurns) == 1) {
                        result = 1; break outer;
                    }
                    if (step == 0) break; // 只处理一次原地
                }
                if (d == 0) continue; // 已处理原地，继续方向
            }
            // 重新实现（更清晰）
            result = solveMove(dp, mr, mc, cr, cc, turn, maxTurns, false);
        } else {
            result = solveMove(dp, mr, mc, cr, cc, turn, maxTurns, true);
        }

        dp[mr][mc][catIdx][turn] = result;
        return result;
    }

    private int solveMove(int[][][][] dp, int mr, int mc, int cr, int cc, int turn, int maxTurns, boolean isCat) {
        int jump = isCat ? catJump : mouseJump;
        // isCat=true: 猫赢=2为好结果; isCat=false: 鼠赢=1为好结果
        int bad = isCat ? 1 : 2; // 对当前方不利的结果
        int good = isCat ? 2 : 1; // 对当前方有利的结果

        int catIdx = cr * cols + cc;

        // 原地不动也是一种选择
        {
            int res;
            if (isCat) {
                res = solveNext(dp, mr, mc, cr, cc, turn, maxTurns, cr, cc);
            } else {
                res = solveNext(dp, mr, mc, cr, cc, turn, maxTurns, mr, mc);
            }
            if (res == good) return good;
        }

        for (int d = 0; d < 4; d++) {
            for (int step = 1; step <= jump; step++) {
                int nr, nc;
                if (isCat) {
                    nr = cr + DR[d] * step;
                    nc = cc + DC[d] * step;
                } else {
                    nr = mr + DR[d] * step;
                    nc = mc + DC[d] * step;
                }
                if (nr < 0 || nr >= rows || nc < 0 || nc >= cols) break;
                if (grid[nr].charAt(nc) == '#') break;

                if (isCat) {
                    if (nr == fr && nc == fc) return good; // 猫到食物，猫赢
                    if (nr == mr && nc == mc) return good; // 猫与鼠同格，猫赢
                    int res = solveNext(dp, mr, mc, cr, cc, turn, maxTurns, nr, nc);
                    if (res == good) return good;
                } else {
                    if (nr == fr && nc == fc) return good; // 鼠到食物，鼠赢
                    if (nr == cr && nc == cc) continue; // 鼠与猫同格，跳过
                    int res = solveNext(dp, mr, mc, cr, cc, turn, maxTurns, nr, nc);
                    if (res == good) return good;
                }
            }
        }
        return bad;
    }

    private int solveNext(int[][][][] dp, int mr, int mc, int cr, int cc, int turn, int maxTurns, int nr, int nc) {
        if ((turn % 2) == 0) {
            // 鼠移动到 (nr,nc)
            int catIdx = cr * cols + cc;
            if (turn + 1 >= maxTurns) return 2;
            int cached = dp[nr][nc][catIdx][turn + 1];
            if (cached != -1) return cached;
            int res = solveMove(dp, nr, nc, cr, cc, turn + 1, maxTurns, true);
            dp[nr][nc][catIdx][turn + 1] = res;
            return res;
        } else {
            // 猫移动到 (nr,nc)
            int catIdx = nr * cols + nc;
            if (turn + 1 >= maxTurns) return 2;
            int cached = dp[mr][mc][catIdx][turn + 1];
            if (cached != -1) return cached;
            int res = solveMove(dp, mr, mc, nr, nc, turn + 1, maxTurns, false);
            dp[mr][mc][catIdx][turn + 1] = res;
            return res;
        }
    }
}
```

> **注意**：上述实现结构复杂，推荐使用更简洁的记忆化递归。核心思路不变：状态 `(鼠位, 猫位, 轮次)`，递归判断当前玩家是否存在必胜走法。

## 复杂度分析

- **时间复杂度**：O(rows^2 * cols^2 * turns * jump)
- **空间复杂度**：O(rows^2 * cols^2 * turns)

## 关键点总结

1. 博弈 DP：鼠回合找任意必胜走法，猫回合找任意必胜走法
2. 状态上限 `rows*cols*2` 防止无限循环
3. 跳跃遇到墙必须停止，不能越墙
