## 题目描述

给定整数数组 `nums`、正整数 `k` 和若干查询 `[l, r, x]`，对每个查询返回区间 `[l, r]` 内乘积对 `k` 取模等于 `x` 的子数组数量。

## 解题思路

线段树，每个节点存储区间信息：`cnt[0..k-1]`（子数组乘积余数分布）、`prefix[0..k-1]`（前缀乘积余数分布）、`suffix[0..k-1]`（后缀乘积余数分布）、`all`（区间整体乘积余数）。合并时跨界子数组 = 左后缀 * 右前缀。

## Java 代码实现

```java
class Solution {
    private int k;
    private long[][] treeCnt, treePre, treeSuf;
    private int[] treeAll;
    private int n;

    public long[] findXValue(int[] nums, int k, int[][] queries) {
        this.k = k;
        n = nums.length;
        treeCnt = new long[4 * n][k];
        treePre = new long[4 * n][k];
        treeSuf = new long[4 * n][k];
        treeAll = new int[4 * n];
        build(nums, 1, 0, n - 1);
        long[] res = new long[queries.length];
        for (int i = 0; i < queries.length; i++) {
            long[] cnt = query(1, 0, n - 1, queries[i][0], queries[i][1]);
            res[i] = cnt[queries[i][2]];
        }
        return res;
    }

    private void build(int[] nums, int node, int s, int e) {
        if (s == e) {
            int r = nums[s] % k;
            treeCnt[node][r] = 1;
            treePre[node][r] = 1;
            treeSuf[node][r] = 1;
            treeAll[node] = r;
            return;
        }
        int mid = (s + e) / 2;
        build(nums, 2*node, s, mid);
        build(nums, 2*node+1, mid+1, e);
        mergeInto(node, 2*node, 2*node+1);
    }

    private void mergeInto(int dst, int L, int R) {
        // cnt = left cnt + right cnt + cross
        for (int i = 0; i < k; i++) treeCnt[dst][i] = treeCnt[L][i] + treeCnt[R][i];
        for (int lr = 0; lr < k; lr++) {
            if (treeSuf[L][lr] == 0) continue;
            for (int rr = 0; rr < k; rr++) {
                if (treePre[R][rr] == 0) continue;
                treeCnt[dst][(lr * rr) % k] += treeSuf[L][lr] * treePre[R][rr];
            }
        }
        // prefix = left prefix + (left all * right prefix)
        for (int i = 0; i < k; i++) treePre[dst][i] = treePre[L][i];
        for (int rr = 0; rr < k; rr++) {
            if (treePre[R][rr] == 0) continue;
            treePre[dst][(treeAll[L] * rr) % k] += treePre[R][rr];
        }
        // suffix = right suffix + (right all * left suffix)
        for (int i = 0; i < k; i++) treeSuf[dst][i] = treeSuf[R][i];
        for (int lr = 0; lr < k; lr++) {
            if (treeSuf[L][lr] == 0) continue;
            treeSuf[dst][(treeAll[R] * lr) % k] += treeSuf[L][lr];
        }
        treeAll[dst] = (treeAll[L] * treeAll[R]) % k;
    }

    // 返回 cnt 数组
    private long[] query(int node, int s, int e, int l, int r) {
        if (l <= s && e <= r) return treeCnt[node];
        int mid = (s + e) / 2;
        if (r <= mid) return query(2*node, s, mid, l, r);
        if (l > mid) return query(2*node+1, mid+1, e, l, r);
        // 合并两侧结果（临时节点）
        long[] Lcnt = query(2*node, s, mid, l, r);
        long[] Rcnt = query(2*node+1, mid+1, e, l, r);
        // 需要前后缀信息，需完整查询
        return queryFull(node, s, e, l, r)[0];
    }

    // 返回 [cnt, pre, suf, all(len-1)]
    private long[][] queryFull(int node, int s, int e, int l, int r) {
        if (l <= s && e <= r) return new long[][]{treeCnt[node], treePre[node], treeSuf[node], new long[]{treeAll[node]}};
        int mid = (s + e) / 2;
        if (r <= mid) return queryFull(2*node, s, mid, l, r);
        if (l > mid) return queryFull(2*node+1, mid+1, e, l, r);
        long[][] left = queryFull(2*node, s, mid, l, r);
        long[][] right = queryFull(2*node+1, mid+1, e, l, r);
        return mergeResults(left, right);
    }

    private long[][] mergeResults(long[][] L, long[][] R) {
        long[] cnt = new long[k], pre = new long[k], suf = new long[k];
        for (int i = 0; i < k; i++) cnt[i] = L[0][i] + R[0][i];
        for (int lr = 0; lr < k; lr++) {
            if (L[2][lr] == 0) continue;
            for (int rr = 0; rr < k; rr++) {
                if (R[1][rr] == 0) continue;
                cnt[(lr * rr) % k] += L[2][lr] * R[1][rr];
            }
        }
        for (int i = 0; i < k; i++) pre[i] = L[1][i];
        for (int rr = 0; rr < k; rr++) {
            if (R[1][rr] == 0) continue;
            pre[((int)L[3][0] * rr) % k] += R[1][rr];
        }
        for (int i = 0; i < k; i++) suf[i] = R[2][i];
        for (int lr = 0; lr < k; lr++) {
            if (L[2][lr] == 0) continue;
            suf[((int)R[3][0] * lr) % k] += L[2][lr];
        }
        long all = (L[3][0] * R[3][0]) % k;
        return new long[][]{cnt, pre, suf, new long[]{all}};
    }
}
```

## 复杂度分析

- 时间复杂度：O((n + q) * k^2 * log n)
- 空间复杂度：O(n * k)

## 关键点总结

1. 线段树节点维护 cnt/prefix/suffix/all_prod 四种信息
2. 合并时跨界子数组 = 左后缀余数 * 右前缀余数
3. 前缀更新：左前缀 + 左整体 * 右前缀
