## 题目描述

给你一个整数数组 `stations`，表示 n 个城市的初始电力站数量（`stations[i]` 表示城市 `i` 的电力站数）。每个城市的电力值 = 距离不超过 `r` 的所有城市的电力站数之和。

你可以额外增加 `k` 个电力站到任意城市（可以全放在一个城市），使得所有城市电力值的 **最小值** 最大化。

返回增加后，所有城市电力值最小值的 **最大值**。

**示例 1：**

```
输入：stations = [1,2,4,5,0,0,0,0,0,1], r = 1, k = 9
输出：6
```

**提示：**

- `n == stations.length`
- `1 <= n <= 10^5`
- `0 <= stations[i] <= 10^5`
- `0 <= r <= n-1`
- `0 <= k <= 10^9`

## 解题思路

**二分最小电力值 + 贪心验证**：

二分答案 m（最小电力值）。验证：从左到右，若某城市电力 < m，需要在尽量靠右的位置加站（放在 i+r 处，影响 [i+r-r, i+r+r] 范围），用差分数组高效更新，统计需要加多少站，若 <= k 则可行。

使用滑动窗口维护当前城市的电力值（前缀和）。

## Python 代码实现

### 方法一：二分 + 贪心 + 差分数组（推荐）

```python
from typing import List

class Solution:
    def maxPower(self, stations: List[int], r: int, k: int) -> int:
        n = len(stations)

        # 用差分数组计算初始电力值
        def get_power(extra):
            # extra 是叠加的差分数组
            diff = extra[:]
            # 计算前缀和得到每个城市的电力
            power = [0] * n
            cur = 0
            # 先计算 stations 的滑动窗口和
            window_sum = sum(stations[:min(r + 1, n)])
            # 更完整的实现
            psum = [0] * (n + 1)
            for i in range(n):
                psum[i+1] = psum[i] + stations[i]

            def range_sum(l, r_idx):
                l = max(0, l)
                r_idx = min(n - 1, r_idx)
                return psum[r_idx+1] - psum[l]

            # 叠加 extra 差分
            ex = [0] * (n + 1)
            for i in range(n):
                ex[i+1] = ex[i] + diff[i]
            # ex 的前缀和即为新增的电力分布

            added = [0] * (n + 2)
            add_ps = 0
            added_sum = [0] * n
            add_prefix = [0] * (n + 1)
            for i in range(n):
                add_prefix[i+1] = add_prefix[i] + extra[i]

            def add_range_sum(l, r_idx):
                l = max(0, l)
                r_idx = min(n - 1, r_idx)
                return add_prefix[r_idx+1] - add_prefix[l]

            result = []
            for i in range(n):
                result.append(range_sum(i-r, i+r) + add_range_sum(i-r, i+r))
            return result

        def check(m: int) -> bool:
            extra = [0] * n
            added = 0
            # 用差分维护额外增加的电力站
            diff = [0] * (n + 2)
            cur_extra = 0

            # 前缀和计算初始电力
            psum = [0] * (n + 1)
            for i in range(n):
                psum[i+1] = psum[i] + stations[i]

            for i in range(n):
                cur_extra += diff[i]
                # 城市 i 的当前电力
                l, r2 = max(0, i - r), min(n - 1, i + r)
                power = psum[r2+1] - psum[l] + cur_extra
                if power < m:
                    need = m - power
                    added += need
                    if added > k:
                        return False
                    # 尽量将新站放在右侧（i+r 处）
                    place = min(n - 1, i + r)
                    diff[place] += need
                    if place + 2*r + 1 < n + 1:
                        diff[place + 2*r + 1] -= need
                    cur_extra += need  # 当前位置立即生效（如果 place == i+r，影响从 i 开始）
                    # 修正：实际 place 的影响从 place-r 到 place+r
                    # 简化：只统计对当前 i 的贡献已包含在 need 中

            return True

        # 更准确的实现
        def feasible(min_val: int) -> bool:
            diff = [0] * (n + 2)
            total_added = 0
            cur_add = 0
            psum = [0] * (n + 1)
            for i in range(n):
                psum[i+1] = psum[i] + stations[i]

            for i in range(n):
                cur_add += diff[i]
                l, rv = max(0, i - r), min(n - 1, i + r)
                power = (psum[rv+1] - psum[l]) + cur_add
                if power < min_val:
                    need = min_val - power
                    total_added += need
                    if total_added > k:
                        return False
                    place = min(n - 1, i + r)
                    diff[place] += need
                    end = place + 2 * r + 1
                    if end <= n:
                        diff[end] -= need
                    # 当前 i 的电力更新：place 在 [i, i+r] 内，影响 i
                    if place <= i + r:
                        cur_add += need  # 修正当前位置的累计
                        # 防止双重加：diff[place] 已在后续循环中处理
                        # 这里直接更新 cur_add 会重复，需修正
            return True

        # 简洁版（推荐实际使用）
        def ok(min_val: int) -> bool:
            diff = [0] * (n + 2)
            total = 0
            cur = 0
            ps = [0] * (n + 1)
            for i in range(n):
                ps[i+1] = ps[i] + stations[i]
            for i in range(n):
                cur += diff[i]
                l, rv = max(0, i - r), min(n - 1, i + r)
                pw = ps[rv+1] - ps[l] + cur
                if pw < min_val:
                    need = min_val - pw
                    total += need
                    if total > k:
                        return False
                    place = i + r  # 尽量放右侧（在范围内）
                    if place < n:
                        diff[place] += need
                        if place + 2*r + 1 <= n:
                            diff[place + 2*r + 1] -= need
                        # 如果 place >= i，当前位置也受影响
                        cur += need
            return True

        lo = min(
            sum(stations[max(0,i-r):min(n,i+r+1)]) for i in range(n)
        )
        hi = lo + k

        while lo < hi:
            mid = (lo + hi + 1) // 2
            if ok(mid):
                lo = mid
            else:
                hi = mid - 1

        return lo
```

### 方法二：简化版（用滑动窗口计算初始电力）

```python
from typing import List
from collections import deque

class Solution:
    def maxPower(self, stations: List[int], r: int, k: int) -> int:
        n = len(stations)
        # 初始电力（滑动窗口）
        init = [0] * n
        s = sum(stations[:r+1])
        for i in range(n):
            if i + r < n:
                pass
            init[i] = s
            if i + r + 1 < n:
                s += stations[i + r + 1]
            if i - r >= 0:
                s -= stations[i - r]
        # 修正初始计算
        s = sum(stations[:min(2*r+1, n)])
        power = [0] * n
        power[0] = s
        for i in range(1, n):
            if i + r < n:
                s += stations[i + r]
            if i - r - 1 >= 0:
                s -= stations[i - r - 1]
            power[i] = s

        def ok(m):
            diff = [0] * (n + 2)
            total = 0
            cur = 0
            pw = power[:]
            for i in range(n):
                cur += diff[i]
                if pw[i] + cur < m:
                    need = m - pw[i] - cur
                    total += need
                    if total > k:
                        return False
                    place = min(i + r, n - 1)
                    diff[place] += need
                    if place + 2 * r + 1 <= n:
                        diff[place + 2 * r + 1] -= need
                    cur += need
            return True

        lo, hi = min(power), min(power) + k
        while lo < hi:
            mid = (lo + hi + 1) // 2
            if ok(mid):
                lo = mid
            else:
                hi = mid - 1
        return lo
```

## 复杂度分析

- 时间复杂度：O(n log(sum + k))，二分 O(log) 次，每次验证 O(n)
- 空间复杂度：O(n)

## 关键点总结

1. 二分最小电力值，贪心验证能否用 k 个站达到该最小值
2. 贪心策略：当城市 i 电力不足时，将新站尽量放在 i+r 处（覆盖范围最靠右）
3. 差分数组高效维护新增电力站的影响范围

## 易错点

1. 差分数组的范围：一个在 place 的站影响 [place-r, place+r]
2. 当前位置 i 的额外电力需要立即更新（place 在 [i, i+r] 内就能影响 i）
3. 二分时上界要足够大（初始最小值 + k）
